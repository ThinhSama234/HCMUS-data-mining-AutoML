\chapter{Phương pháp đề xuất}

\section{Tổng quan}
\label{sec:tongquan-pp}
Đồ án theo \textbf{Hướng 1 --- trình bày lại và tái thực nghiệm}: nhóm kế thừa
giao thức đánh giá công bằng của AMLB~\cite{gijsbers2024amlb} làm nền tảng, đồng
thời đóng góp một hệ thống \textit{AutoML Bench Console} giúp quy trình đánh giá
trở nên dễ tái lập và thuận tiện trên dữ liệu thực. Phương pháp gồm hai phần bổ
trợ nhau: (i) \emph{giao thức đánh giá công bằng} bảo đảm tính đúng đắn của phép
so sánh (Mục~\ref{sec:giaothuc}), và (ii) \emph{kiến trúc hệ thống} hiện thực hoá
giao thức đó theo hướng tự động và tái lập (Mục~\ref{sec:kientruc}).

\section{Giao thức đánh giá công bằng}
\label{sec:giaothuc}
Nguyên tắc trung tâm là \emph{mọi thư viện phải được đánh giá trong cùng một điều
kiện}. Cụ thể, với mỗi bộ dữ liệu $D$, giao thức cố định bốn yếu tố:
\begin{enumerate}
    \item \textbf{Cách chia fold:} cùng một phân hoạch $K$ fold cho mọi thư viện.
    \item \textbf{Ngân sách thời gian:} cùng thời gian tối đa trên mỗi fold.
    \item \textbf{Độ đo:} chọn theo loại tác vụ (AUC cho phân lớp nhị phân,
    log-loss cho phân lớp đa lớp, RMSE cho hồi quy).
    \item \textbf{Tài nguyên:} cùng số nhân CPU và bộ nhớ, được bảo đảm bằng môi
    trường Docker cô lập.
\end{enumerate}
Ngoài ra, mỗi lần chạy được ghi nhận trạng thái (\texttt{done}/\texttt{failed})
kèm nguyên nhân, để các thất bại được phân tích thay vì bị bỏ qua.

\paragraph{Xếp hạng theo thứ hạng trung bình.} Sau khi thu điểm số của $k$ thư
viện trên $m$ bộ dữ liệu, ta chuẩn hoá về chiều ``lớn hơn là tốt hơn'' (lấy dấu
âm với log-loss và RMSE), rồi trên mỗi bộ dữ liệu $D_j$ gán cho thư viện $F_i$ một
thứ hạng $r_{ij}$ ($1$ là tốt nhất). Thứ hạng trung bình của thư viện $F_i$ là
\[
    \bar{r}_i = \frac{1}{m}\sum_{j=1}^{m} r_{ij},
\]
và các thư viện được xếp hạng chung theo $\bar{r}_i$ tăng dần. Thứ hạng trung
bình được ưa dùng vì ít nhạy với sự khác biệt thang đo giữa các bộ dữ liệu.

\section{Thuật toán quy trình đánh giá}
\label{sec:thuattoan}
Thuật toán~\ref{alg:bench} tóm tắt quy trình đánh giá cho một lần chạy gồm tập
thư viện $\mathcal{F}$ và tập dữ liệu $\mathcal{D}$ dưới một ràng buộc tài nguyên
$c$.

\begin{algorithm}[H]
\caption{Quy trình đánh giá AutoML công bằng}
\label{alg:bench}
\begin{algorithmic}[1]
\Require Tập thư viện $\mathcal{F}$, tập dữ liệu $\mathcal{D}$, ràng buộc $c$ (fold, thời gian, tài nguyên)
\Ensure Bảng điểm $\mathcal{S}$ và thứ hạng trung bình $\{\bar{r}_i\}$
\State $\mathcal{S} \gets \emptyset$
\ForAll{$D_j \in \mathcal{D}$}
    \State cố định phân hoạch fold của $D_j$ theo $c$
    \ForAll{$F_i \in \mathcal{F}$}
        \State chạy $F_i$ trên $D_j$ trong Docker với ràng buộc $c$
        \If{chạy thành công}
            \State chấm điểm dự đoán theo độ đo của $D_j$; lưu điểm và chi phí vào $\mathcal{S}$
        \Else
            \State ghi nhận trạng thái \texttt{failed} kèm nguyên nhân
        \EndIf
    \EndFor
\EndFor
\State chuẩn hoá điểm về chiều ``lớn hơn là tốt hơn''; tính thứ hạng $r_{ij}$ trên mỗi $D_j$
\State $\bar{r}_i \gets \frac{1}{m}\sum_{j} r_{ij}$ cho mỗi $F_i$
\State \Return $\mathcal{S}$, $\{\bar{r}_i\}$
\end{algorithmic}
\end{algorithm}

\noindent Trên mỗi bộ dữ liệu, sau khi chuẩn hoá về chiều ``lớn hơn là tốt hơn''
(với RMSE và log-loss ta xếp giá trị nhỏ hơn lên trước), thứ hạng $r_{ij}$ là:
\begin{itemize}
    \item breast\_cancer: H2O $=1$, AutoGluon $=2$, FLAML $=3$.
    \item california\_housing: AutoGluon $=1$, H2O $=2$, FLAML $=3$.
    \item wine: H2O $=1$, AutoGluon $=2$, FLAML $=3$.
\end{itemize}
Thứ hạng trung bình:
\[
    \bar{r}_{\text{H2O}} = \tfrac{1+2+1}{3} = 1.33,\quad
    \bar{r}_{\text{AutoGluon}} = \tfrac{2+1+2}{3} = 1.67,\quad
    \bar{r}_{\text{FLAML}} = \tfrac{3+3+3}{3} = 3.00.
\]
Như vậy thứ tự xếp hạng là H2O $\succ$ AutoGluon $\succ$ FLAML. Ví dụ cho thấy cơ
chế xếp hạng tổng hợp được nhiều loại độ đo khác nhau về một thang so sánh thống
nhất, đồng thời làm rõ vì sao cần chuẩn hoá chiều tối ưu trước khi gán thứ hạng.
\end{vidu}

\section{Phân tích độ phức tạp}
\label{sec:dophuctap}
Gọi $k = |\mathcal{F}|$, $m = |\mathcal{D}|$ và $K$ là số fold. Quy trình ở
Thuật toán~\ref{alg:bench} thực hiện $k\cdot m\cdot K$ lần huấn luyện--đánh giá
độc lập; nếu mỗi lần bị chặn bởi ngân sách thời gian $B$ thì tổng thời gian chạy
là $O(k\, m\, K\, B)$. Bước tính thứ hạng có chi phí $O\!\left(m\, k\log k\right)$
do phải sắp xếp $k$ thư viện trên mỗi trong $m$ bộ dữ liệu, không đáng kể so với
chi phí huấn luyện. Vì các lần chạy độc lập, quy trình có tính song song cao: có
thể phân tán theo (thư viện $\times$ bộ dữ liệu) để rút ngắn thời gian tường minh.

\section{Thảo luận thiết kế}
\label{sec:thaoluan-pp}
Việc tách bạch \emph{giao thức} khỏi \emph{hiện thực} mang lại hai lợi ích. Thứ
nhất, dùng Docker bảo đảm mỗi thư viện chạy trong môi trường đồng nhất và cô lập,
loại bỏ sai khác do phiên bản thư viện phụ thuộc --- một nguyên nhân phổ biến gây
ra so sánh không công bằng. Thứ hai, việc ghi nhận thất bại kèm nguyên nhân biến
các lỗi thành dữ liệu phân tích (xem Mục~\ref{sec:phantichloi}), thay vì để chúng
âm thầm làm sai lệch kết quả. Đánh đổi của thiết kế là chi phí hạ tầng (Docker,
PostgreSQL, MinIO) và yêu cầu môi trường Linux/đám mây cho các thư viện nặng.
